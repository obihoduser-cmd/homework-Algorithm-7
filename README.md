#include <iostream>
#include <vector>
#include <string>
#include <algorithm>

using namespace std;

// 1. БЫСТРАЯ СОРТИРОВКА (QUICK SORT)

template <typename T>
void quickSort(vector<T>& arr, int low, int high) {
    if (low >= high) {
        return;
    }

    // Выбираем средний элемент в качестве опорного
    T pivot = arr[low + (high - low) / 2];

    int i = low;
    int j = high;

    while (i <= j) {
        while (arr[i] < pivot) {
            i++;
        }

        while (arr[j] > pivot) {
            j--;
        }

        if (i <= j) {
            swap(arr[i], arr[j]);
            i++;
            j--;
        }
    }

    if (low < j) {
        quickSort(arr, low, j);
    }

    if (i < high) {
        quickSort(arr, i, high);
    }
}

// 2. СОРТИРОВКА СЛИЯНИЕМ (MERGE SORT)

template <typename T>
void merge(vector<T>& arr, vector<T>& temp,
           int left, int mid, int right) {

    int i = left;
    int j = mid + 1;
    int k = left;

    // Сравниваем элементы двух частей
    while (i <= mid && j <= right) {
        if (arr[i] <= arr[j]) {
            temp[k++] = arr[i++];
        }
        else {
            temp[k++] = arr[j++];
        }
    }

    // Копируем оставшиеся элементы
    while (i <= mid) {
        temp[k++] = arr[i++];
    }

    while (j <= right) {
        temp[k++] = arr[j++];
    }

    // Возвращаем отсортированную часть в исходный массив
    for (int p = left; p <= right; p++) {
        arr[p] = temp[p];
    }
}

template <typename T>
void mergeSort(vector<T>& arr, vector<T>& temp,
               int left, int right) {

    if (left >= right) {
        return;
    }

    int mid = left + (right - left) / 2;

    mergeSort(arr, temp, left, mid);
    mergeSort(arr, temp, mid + 1, right);

    merge(arr, temp, left, mid, right);
}

// 3. ПИРАМИДАЛЬНАЯ СОРТИРОВКА (HEAP SORT)

// Восстановление свойств max-heap
template <typename T>
void heapify(vector<T>& arr, int n, int i) {

    while (true) {
        int largest = i;

        int left = 2 * i + 1;
        int right = 2 * i + 2;

        if (left < n && arr[left] > arr[largest]) {
            largest = left;
        }

        if (right < n && arr[right] > arr[largest]) {
            largest = right;
        }

        if (largest == i) {
            break;
        }

        swap(arr[i], arr[largest]);

        i = largest;
    }
}

template <typename T>
void heapSort(vector<T>& arr) {

    int n = arr.size();

    // Построение max-heap
    for (int i = n / 2 - 1; i >= 0; i--) {
        heapify(arr, n, i);
    }

    // Последовательно переносим максимум в конец
    for (int i = n - 1; i > 0; i--) {

        swap(arr[0], arr[i]);

        heapify(arr, i, 0);
    }
}

// Проверка правильности сортировки
template <typename T>
bool isSorted(const vector<T>& arr) {

    for (int i = 1; i < arr.size(); i++) {
        if (arr[i] < arr[i - 1]) {
            return false;
        }
    }

    return true;
}

Контрольные вопросы:
1. Какова сложность быстрой сортировки в лучшем, среднем и худшем случае?
- Лучший случай — O(n log n).
- Средний случай — O(n log n).
- Худший случай — O(n²).
2. Когда возникает худший случай для быстрой сортировки?

Когда опорный элемент постоянно оказывается минимальным или максимальным, из-за чего массив делится на две неравные части: одна содержит n − 1 элементов, другая — 0. Например, такое возможно при сортировке уже упорядоченного массива с выбором последнего элемента в качестве опорного.

3. Как можно улучшить быструю сортировку для защиты от худшего случая?

Использовать случайный выбор опорного элемента, медиану трёх элементов или переключаться на другой алгоритм сортировки при слишком глубокой рекурсии.

4. Какова сложность сортировки слиянием? Почему она всегда O(n log n)?

Временная сложность — O(n log n) в лучшем, среднем и худшем случаях. Массив каждый раз делится примерно пополам, образуя log n уровней. На каждом уровне выполняется объединение элементов общей сложностью O(n).

5. Какие преимущества и недостатки имеет сортировка слиянием перед быстрой?

Преимущества:
- Стабильная сложность O(n log n).
- Устойчивая сортировка.
- Хорошо подходит для больших объёмов данных и связных списков.
Недостатки:
- Требует дополнительную память O(n) для массивов.
- Может работать медленнее быстрой сортировки на практике из-за дополнительных операций копирования и выделения памяти.
6. Как работает пирамидальная сортировка? Какова её сложность?

Сначала строится бинарная куча — структура, в которой родительский элемент больше своих детей (для сортировки по возрастанию). Затем корневой элемент меняется местами с последним элементом кучи, размер кучи уменьшается, а её свойства восстанавливаются.

Временная сложность во всех случаях — O(n log n), дополнительная память — O(1) для итеративной реализации.

7. Какая из трёх сортировок устойчивая? Почему?

Сортировка слиянием является устойчивой при стандартной реализации, если при равенстве элементов первым выбирается элемент из левой части. Это означает, что относительный порядок одинаковых элементов сохраняется.

Быстрая и пирамидальная сортировки в стандартной реализации неустойчивы.

8. Какая из трёх сортировок требует дополнительной памяти? Сколько?
- Быстрая сортировка — в среднем O(log n) для стека рекурсии, в худшем случае O(n).
- Сортировка слиянием — O(n) для временного массива.
- Пирамидальная сортировка — O(1) при итеративной реализации.
9. Какая из трёх сортировок лучше работает на почти отсортированных данных?

Из этих трёх алгоритмов быстрая сортировка с хорошим выбором опорного элемента обычно работает эффективно. Однако ни один из трёх алгоритмов не гарантирует линейное время на почти отсортированных данных. Сортировка вставками для такого случая обычно подходит лучше.

10. Зачем в пирамидальной сортировке сначала строится куча?

Чтобы организовать элементы по приоритету. Благодаря этому максимальный элемент находится в корне и его можно быстро переместить в конец массива. После каждого извлечения куча восстанавливается.

11. Что такое «разделяй и властвуй»? Как он применяется в быстрой сортировке и сортировке слиянием?

«Разделяй и властвуй» — принцип решения задачи путём разделения её на более мелкие подзадачи, решения этих подзадач и объединения результатов.
- В быстрой сортировке массив разделяется относительно опорного элемента, после чего части сортируются отдельно.
- В сортировке слиянием массив делится на половины, которые сортируются отдельно, а затем объединяются в один упорядоченный массив.
12. Можно ли реализовать быструю сортировку итеративно (без рекурсии)?

Да. Вместо рекурсивных вызовов используется собственный стек, в который записываются границы частей массива, требующих сортировки. Затем эти части обрабатываются в цикле.

13. Почему быстрая сортировка обычно быстрее сортировки слиянием на практике?

Она обычно выполняет меньше дополнительных операций с памятью, сортирует элементы непосредственно в исходном массиве и обладает хорошей локальностью обращений к памяти. Поэтому часто показывает более высокую практическую производительность.

14. Когда выгодно использовать пирамидальную сортировку?

Когда важны гарантированная сложность O(n log n) и минимальное использование дополнительной памяти. Она подходит для систем с ограниченными ресурсами.

15. Какой алгоритм сортировки вы бы выбрали для встроенных систем с ограниченной памятью?

Пирамидальную сортировку, поскольку она работает за O(n log n) и требует O(1) дополнительной памяти при итеративной реализации. Это позволяет сортировать данные без выделения дополнительного массива.
